async getAll(req, res) {
        // Извлекаем параметры фильтрации из запроса
        let { id_category, price_min, price_max, rating_min, specifications, limit, page } = req.query;
        console.log("Параметры запроса:", req.query);
        // Устанавливаем значения по умолчанию для страницы и лимита, если они не были предоставлены
        page = page || 1;
        limit = limit || 9;
        let offset = page * limit - limit; // Вычисляем смещение для пагинации

        let filters = {}; // Объект для фильтрации товаров по данной категории, цене и рейтингу
        let specificationFilters = []; // Массив условий для фильтрации товаров по их спецификациям

        // Обрабатываем фильтр по категории
        if (id_category) {
            filters.id_category = id_category; // Добавляем фильтр по id категории
        }
        // Обрабатываем фильтры по ценовому диапазону
        if (price_min || price_max) {
            filters.price_product = {};
            if (price_min) filters.price_product[Op.gte] = price_min; // Добавляем минимальную цену
            if (price_max) filters.price_product[Op.lte] = price_max; // Добавляем максимальную цену
        }
        // Обрабатываем фильтр по минимальному рейтингу
        if (rating_min) {
            filters.rating = { [Op.gte]: rating_min }; // Добавляем фильтр по рейтингу
        }

        // Обрабатываем фильтр по спецификациям
        // if (specifications) {
        //     specifications = JSON.parse(specifications); // Преобразуем строку в формате JSON со списком спецификаций и их значений
        //     console.debug(specifications)
        //     for (const [name, value] of Object.entries(specifications)) {
        //         // Для каждого значения создаём условие фильтрации и добавляем его в массив
        //         specificationFilters.push({
        //             '$specifications.attribute.name_attribute$': name, // Используем имя атрибута
        //             '$specifications.value_specification$': value // Используем значение спецификации
        //         });
        //     }
        // }
        if (specifications && Array.isArray(specifications)) {
            // Проходимся по каждой спецификации
            for (const spec of specifications) {
                console.log(spec)
                // Проверяем, есть ли имя атрибута и значение характеристики
                if (spec.id_attribute && spec.value_specification) {
                    // Добавляем фильтр по атрибуту и его значению
                    specificationFilters.push({
                        '$specifications.attribute.id_attribute$': spec.id_attribute,
                        '$specifications.value_specification$': spec.value_specification
                    });
                }
            }
        }

        if (specificationFilters.length > 0) {
            filters[Op.and] = specificationFilters;
        }
        // Пытаемся получить список товаров, учитывая заданные фильтры
        try {
            let products = await Product.findAndCountAll({
                where: { ...filters }, // Используем объект filters для фильтрации
                include: [
                    {
                        model: Specification,
                        as: 'specifications',
                        // required: specificationFilters.length > 0,
                        // where: specificationFilters.length > 0 ? { [Op.and]: specificationFilters } : undefined,
                        include: {
                            model: Attribute,
                            as: 'attribute',
                            include: {
                                model: Attribute_group,
                                as: 'attribute_group',
                            }
                        }
                    }
                ],
                limit, // Ограничение количества товаров в ответе
                offset, // Смещение для пагинации
                distinct: true, // Это нужно для корректного подсчёта общего количества товаров
                // ...(specificationFilters.length > 0 && { where: { [Op.and]: specificationFilters } }) // Условия для фильтрации по спецификациям
            });
            // Возвращаем список товаров клиенту
            return res.json(products);
        } catch (e) {
            console.error(e)
            return res.status(500).json({ message: "Ошибка при получении списка товаров:" + e.message });
        }
    }
